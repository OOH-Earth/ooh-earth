import { test } from '@playwright/test';
import { mockBase44 } from './fixtures/mockBase44';
test.use({ viewport: { width: 1280, height: 800 } });
test('claim', async ({ page }) => {
  test.setTimeout(90000);
  await page.route('**/api/**', (r) => r.fulfill({ status: 404, contentType: 'application/json', body: '{}' }));
  const locs: Record<string, unknown> = {};
  for (let i = 0; i < 6; i++) locs['l' + i] = { id: 'l' + i, title: `Spot ${i}`, type: 'billboard', address: `${i} Field Way`, lat: 13.75 + i * 0.001, lng: 100.5 + i * 0.001, status: 'pending', created_date: '2020-01-01T00:00:00.000Z' };
  await mockBase44(page, { user: null, locations: locs, fieldChecks: {} });
  await page.addInitScript(() => localStorage.setItem('ooh_nudge_map_report', '1'));
  await page.goto('/adbusting'); await page.getByText('Spot 1').first().waitFor({ timeout: 30000 });
  const claim = page.getByRole('button', { name: 'Claim', exact: true }).first();
  const b = (await claim.boundingBox())!;
  console.log('claim box', JSON.stringify(b), 'count', await page.getByRole('button', { name: 'Claim', exact: true }).count());
  console.log('at centre:', await page.evaluate(([x, y]) => { const e = document.elementFromPoint(x, y)!; return `${e.tagName} ${String(e.className).slice(0, 80)} text="${(e as HTMLElement).innerText?.slice(0, 20)}"`; }, [b.x + b.width / 2, b.y + b.height / 2]));
  await claim.click(); await page.waitForTimeout(1200);
  console.log('dialogs after click:', await page.getByRole('dialog').count(), '| body text has claim title:', await page.getByText(/claim/i).count());
  await page.screenshot({ path: '/tmp/claude-1000/-home-hiker123-oohearth/4ae9710b-6b4e-4b26-828a-4f3dc6ee6265/scratchpad/adb-claim.png' });
});
